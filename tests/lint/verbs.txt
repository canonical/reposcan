[stative-predicates]
# Third-person forms of stative verbs. The subject is elided, but no action reading is
# possible -- `is_sqlite(data)` cannot mean "make this sqlite" -- so the argument is
# unambiguously the subject. Allowed anywhere, including free functions.
# (`succeeded` is past tense rather than third-person present, but reads the same way:
# "did this succeed?", never "succeed this".)
can
has
is
knows
should
succeeded

[dynamic-predicates]
# Third-person forms of dynamic verbs. `run.matches(other)` is a complete sentence
# because `self` is the subject. As a free function the subject is elided and the name
# collides with the base form: `returns_a_failure(x)` could just as well mean the
# function returns one. Instance methods only.
builds
converts
holds
ignores
matches
names
returns
